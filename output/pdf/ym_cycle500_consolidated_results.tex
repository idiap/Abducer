\chapter*{Abstract}
\addcontentsline{toc}{chapter}{Abstract}
This monograph consolidates the reviewed mathematical output of a long-term
Yang--Mills mass-gap research programme. Its strongest positive results are:
(i) a dense-core spectral criterion converting a common diagonal semigroup
rate into a gap; (ii) a finite stationary Perron--Doob/OS construction whose
OS space is $L^2(\pi)$ and whose descended time translation is a positive
self-adjoint Markov contraction; (iii) an exact source-visible edge-resolvent
threshold; (iv) a positive-block shorting theorem showing that an upper WTR
bound is structural at an already known global edge; and (v) an exact
candidate-edge/shorted-Dirichlet-coercivity equivalence. The programme also
proved sharp contour-incidence bounds, finite graph completeness statements,
many exact $SU(2)$ coefficient identities, and a family of scoped no-go
theorems for finite-complexity carriers.

None of these results locates a physical four-dimensional Wilson spectral
edge, proves a lower signed connected response at that edge, derives a
surviving transmuted Wilson coefficient, or completes continuum OS survival.
Those four items form the exact remaining bridge for the selected route.

\section{Operators, spectral measures, and OS notation}
Let $H\geq0$ be self-adjoint on a Hilbert space $\mathcal K$, with semigroup
$e^{-tH}$. For $\psi\in\mathcal K$, write
\[
 L_\psi(t)=\langle\psi,e^{-tH}\psi\rangle
 =\int_{[0,\infty)}e^{-t\lambda}\,d\mu_\psi(\lambda).
\]
In an OS reconstruction, $\Omega$ denotes the vacuum and $q(F)$ the OS class
of a positive-time observable. Pointed centering means
$(I-P_\Omega)q(F)$; full-ground-space centering means $(I-P_0)q(F)$.

For a retained/eliminated decomposition $P+R=I$, write a self-adjoint block
operator as
\[
B=\begin{pmatrix}A&C\\D&E\end{pmatrix},\qquad D=C^*.
\]
At a proposed order bound $q$, set $X=qP-A$ and $H_R=qR-E$. The shorted
operator is $S_P(q)=X-CH_R^{-1}D$ in the strict-tail regime.

For a source $s$ coupled by $D$ to a tail operator $E$ with upper edge $e^+$,
the source-visible edge measure is
\[
 \nu_s(\Lambda)=\|\one_\Lambda(E)Ds\|^2,\qquad
 F_s(u)=\nu_s([e^+-u,e^+]).
\]

\chapter{The dense-core spectral theorem}

\begin{theorem}[Vector spectral criterion, VSC-1]\label{thm:vsc}
Let $H\geq0$ be self-adjoint and $m>0$. For every $\psi\in\mathcal K$,
the following are equivalent:
\begin{enumerate}
\item there are finite $C_\psi,T_\psi$ with
$L_\psi(t)\leq C_\psi e^{-mt}$ for $t\geq T_\psi$;
\item $\mu_\psi([0,m))=0$;
\item $L_\psi(t)\leq\|\psi\|^2e^{-mt}$ for every $t\geq0$.
\end{enumerate}
For $\psi\neq0$ these are also equivalent to
$\liminf_{t\to\infty}-t^{-1}\log L_\psi(t)\geq m$.
\end{theorem}
\begin{proof}
If the spectral measure is supported in $[m,\infty)$, the all-time bound is
immediate from the spectral integral. Conversely, suppose the eventual bound
holds and $\mu_\psi([0,m))>0$. Then for some $\varepsilon>0$ the interval
$[0,m-\varepsilon]$ has positive mass $a$. Hence
$L_\psi(t)\geq a e^{-(m-\varepsilon)t}$, contradicting
$C_\psi e^{-mt}$ as $t\to\infty$. The logarithmic formulation follows by
taking logarithms of positive Laplace transforms. The zero vector is handled
separately and creates no logarithmic expression.
\end{proof}

\begin{theorem}[Dense-core gap criteria, DCG-1 and DCG-2]\label{thm:dcg}
Let $\mathcal K_0=(\ker H)^\perp$ and let $S\subset\mathcal K_0$ be total.
There is one $m>0$ such that every $\psi\in S$ obeys the eventual bound of
Theorem~\ref{thm:vsc}, with vector-dependent finite constants, if and only if
$\spec(H|_{\mathcal K_0})\subset[m,\infty)$.

If $H\Omega=0$ and $S\subset\Omega^\perp$ is total in $\Omega^\perp$, the
same premise implies
\[
 \ker H=\mathbb C\Omega,\qquad
 \spec(H)\subset\{0\}\cup[m,\infty).
\]
\end{theorem}
\begin{proof}
Theorem~\ref{thm:vsc} places every vector of $S$ in the closed spectral
subspace $E_H([m,\infty))\mathcal K$. Totality places all of $\mathcal K_0$
there. The converse is the sharp all-time spectral bound. In the pointed
branch, totality in $\Omega^\perp$ rules out any additional zero-energy vector,
because such a vector is orthogonal to the closed positive spectral subspace.
\end{proof}

\begin{corollary}[Conditional OS handoff, OS-DCG]
In an OS theory, suppose the centered classes $(I-P_\Omega)q(F_\lambda)$ from
an admissible positive-time core are total in $\Omega^\perp$ and their exact
connected diagonal coefficients equal
$\langle\psi_\lambda,e^{-tH}\psi_\lambda\rangle$. A common physical exponent
$m$ with label-dependent finite constants proves vacuum uniqueness and a gap
at least $m$. With $(I-P_0)$ centering and totality only in $(\ker H)^\perp$,
one proves a gap above the full ground space but not uniqueness.
\end{corollary}
\begin{remark}
The unproved Yang--Mills inputs are the continuum OS construction, a fixed
physical-time cutoff handoff on one cofinal sequence, a common Wilson rate,
nonnull descent, and post-quotient totality. No generator-domain or
unit-sphere-uniform constant is required once these inputs are present.
\end{remark}

\chapter{Observable completeness and hidden sectors}

\begin{theorem}[Loop sigma-completeness criterion]\label{thm:loopsigma}
Let $(z_n)$ be bounded countably many Wilson variables and let $\Sigma_W$ be
their completed sigma-algebra. Rational star-polynomials in the $z_n$ are
dense in $L^2(\Sigma_W)$. Consequently they are dense in the full positive-time
gauge-invariant $L^2$ space exactly when
$\Sigma_W=\Sigma_+^{\rm gi}$ modulo null sets.
\end{theorem}
\begin{proof}
The joint map $Z$ takes values in a compact metrizable product. On its compact
support, cylinder star-polynomials separate points and contain constants.
Stone--Weierstrass gives uniform density in continuous functions; continuous
functions are dense in $L^2$ of the push-forward measure. Pullback gives the
claim, and rational coefficients are dense in the complex coefficients.
\end{proof}

\begin{proposition}[OS quotient continuity]
For reflection-invariant measure and reflection-positive form,
$\|q(F)\|_{\rm OS}^2\leq\|F\|_2^2$. Hence raw $L^2$ density of bounded loop
representatives descends to OS density, provided the full OS algebra and
measure have already been constructed.
\end{proposition}
\begin{proof}
Cauchy--Schwarz and reflection invariance give
$|\int\overline{\theta F}F,d\mu|\leq\|\theta F\|_2\|F\|_2=\|F\|_2^2$.
Thus the quotient map is continuous and carries a dense raw core to a dense
quotient core.
\end{proof}

\begin{counterexample}[Hidden nonnull sector]
Take $X=X_d\times\{\pm1\}$, let all Wilson loops depend only on $X_d$, and use
identity or tensor reflection. The function $h(x,s)=s$ is nonzero, nonnull,
and orthogonal to every loop polynomial. Thus finite-graph loop completeness
and reflection positivity do not imply full continuum OS cyclicity.
\end{counterexample}

\begin{theorem}[JCT-FR]\label{thm:jct}
Let $X_1,\ldots,X_r$ be compatible repair contours with one baseline and
linear repair space. Put $V=\bigcup_iX_i$ and
$D_\varepsilon=\triangle_{i:\varepsilon_i=1}X_i$ for
$0\neq\varepsilon\in\mathbb F_2^r$. Then for some $\varepsilon$,
\[
 |D_\varepsilon|\geq\alpha_r|V|,
 \qquad \alpha_r=\frac{2^{r-1}}{2^r-1}.
\]
The constant is sharp using only incidence information. If the incidence rank
is $d$, replace $r$ by $d$.
\end{theorem}
\begin{proof}
At each face in $V$, the nonzero incidence functional equals one for exactly
$2^{r-1}$ of the $2^r-1$ nonzero vectors $\varepsilon$. Summing
$|D_\varepsilon|$ over nonzero $\varepsilon$ therefore gives
$2^{r-1}|V|$. At least one term is no smaller than the average. Sharpness is
realized by the abstract incidence system $V=\mathbb F_2^r\setminus\{0\}$,
$X_i=\{v:v_i=1\}$, in which every nonzero $\varepsilon$ attains the average.
\end{proof}
\begin{remark}
The resulting Peierls factor is
$\exp[-2\beta\cos\rho\,\alpha_r|V|]$. Connected Wilson realization,
source coefficients, and multiplicity entropy are not supplied by the theorem.
\end{remark}

\chapter{Spectral ratios and source-visible edges}

\begin{theorem}[FCC spectral-ratio theorem]\label{thm:fcc}
Let $T=e^{-RH}$, $R>0$, and $\psi\neq0$. Define
$r_n=\|T^{n+1}\psi\|/\|T^n\psi\|$. Then $r_n$ is nondecreasing and
\[
 \lim_{n\to\infty}r_n=e^{-R\inf\supp\mu_\psi}.
\]
An eventual bound $r_n\leq q<1$ is equivalent to exclusion of cyclic spectral
support below $-R^{-1}\log q$.
\end{theorem}
\begin{proof}
The sequence $a_n=\|T^n\psi\|^2$ is a Hausdorff moment sequence.
Cauchy--Schwarz gives $a_{n+1}^2\leq a_na_{n+2}$, so
$r_n^2=a_{n+1}/a_n$ is nondecreasing. The $n$th-root principle for positive
moments identifies the limit with the top of the support of $e^{-2R\lambda}$,
which is $e^{-2R\inf\supp\mu_\psi}$ after taking square roots.
\end{proof}
\begin{counterexample}[No finite bootstrap]
Put arbitrarily small positive spectral mass at an arbitrarily soft edge. Any
fixed finite list of ratios can look favorable while the monotone sequence
later rises to one. Thus finite favorable ratios do not imply a gap.
\end{counterexample}

\begin{theorem}[Source-visible WTR threshold]\label{thm:wtr}
Let $z=e^++\delta$, $\delta>0$. Then
\[
 \langle s,D^*(z-E)^{-1}Ds\rangle
 =\int_{[0,\infty)}\frac{dF_s(x)}{\delta+x}.
\]
Assume $\|D\|\leq M$ and, near zero, $F_s(u)\leq Cu^\gamma$ uniformly.
Then the resolvent is $O(1)$ for $\gamma>1$,
$O(\log(1/\delta))$ for $\gamma=1$, and
$O(\delta^{\gamma-1})$ for $0<\gamma<1$. An edge atom of mass $a$ contributes
$a/\delta$. Matching lower mass laws give matching sharp orders.
\end{theorem}
\begin{proof}
Push the spectral measure forward by $x=e^+-\lambda$. Separate a fixed far
part, bounded by $M^2$, from $x\in[0,1]$. Stieltjes integration by parts gives
\[
 \int_{[0,1]}\frac{dF(x)}{\delta+x}
 \leq C_0+C\int_0^1\frac{x^\gamma}{(\delta+x)^2}\,dx.
\]
The substitution $x=\delta y$ yields the three regimes. An atom at $x=0$
appears exactly as $a/\delta$. Reversing the inequalities under a matching
lower law proves sharpness.
\end{proof}
\begin{remark}
An upper law gives upper growth only. It does not locate the Wilson edge or
prove the lower signed connected window mass required for a physical response.
\end{remark}

\chapter{The finite stationary Wilson--OS theorem}

\begin{theorem}[Perron--Doob stationary OS construction]\label{thm:stationaryOS}
Fix a registered finite spatial $SU(2)$ lattice and $\beta>0$. Let $K$ be a
continuous positivity-preserving, positive-semidefinite, self-adjoint,
injective Wilson slab operator with Perron pair $(\lambda_0,h)$, $h>0$,
normalized by $\int h^2\,d\mu=1$. Define
\[
 P(x,dy)=\frac{K(x,y)h(y)}{\lambda_0h(x)}\,\mu(dy),
 \qquad \pi(dx)=h(x)^2\mu(dx),
\]
and $T=\lambda_0^{-1}h^{-1}Kh$ on $L^2(\pi)$. The stationary bi-infinite
Markov chain is reflection positive across a bond. Its completed future OS
space is unitarily isomorphic to $L^2(\pi)$, and time translation descends to
the positive self-adjoint Markov contraction $T$. Every nonzero bounded
one-slice gauge-invariant source survives the OS quotient.
\end{theorem}
\begin{proof}
The Doob formula and $Kh=\lambda_0h$ show that $P$ is Markov. Self-adjointness
of $K$ gives detailed balance with $\pi$, hence a reversible stationary chain
by Kolmogorov extension. For a future observable $F$, set
$CF=\mathbb E[F\mid X_1]$. The Markov property and reflection give
\[
 \langle q(F),q(G)\rangle_{\rm OS}=\langle CF,T\,CG\rangle_{L^2(\pi)}.
\]
Because $T\geq0$, the map $JF=T^{1/2}CF$ realizes the quotient norm and has
kernel exactly the OS null space. Injectivity of the Wilson slab follows from
strictly positive Peter--Weyl multipliers; hence $T^{1/2}$ has dense range and
$J$ completes onto $L^2(\pi)$. Conditional expectation intertwines one time
shift with $T$, so the descended shift is the positive self-adjoint Markov
contraction. Finally, for nonzero one-slice $f$, injectivity and positivity give
$\langle f,Tf\rangle>0$, proving survival.
\end{proof}
\begin{remark}
The stationary endpoint law $h^2\mu$ is a newly registered finite object and
is not automatically the frozen open-slab law. Compactness gives finite-volume
spectral discreteness, but no regulator-uniform physical edge, continuum
semigroup, Wilson-core totality, or mass gap.
\end{remark}

\chapter{Shorting theory and the exact candidate-edge test}

\begin{theorem}[GEWTR-1: positive-block shortening]\label{thm:gewtr}
Let $B=\bigl(\begin{smallmatrix}A&C\\D&E\end{smallmatrix}\bigr)$ be bounded
self-adjoint and let $\lambda\geq\sup\spec(B)$. Put
$X=\lambda P-A$ and $H_R=\lambda R-E$. Then there is a contraction $K$ with
\[
 C=X^{1/2}KH_R^{1/2},\qquad D=H_R^{1/2}K^*X^{1/2}.
\]
For every retained $s$, $Ds\perp\ker H_R$ and
\[
 \|H_{R,\rm red}^{-1/2}Ds\|^2\leq\langle s,Xs\rangle,
\quad
 \langle s,D^*(H_R+\delta)^{-1}Ds\rangle
 \leq\langle s,Xs\rangle.
\]
\end{theorem}
\begin{proof}
The block $\lambda I-B$ is positive. The standard positive-block
factorization supplies the contraction $K$ without requiring closed range.
Substituting the factorization into the quadratic form gives the reduced
inverse estimate; the regularized resolvent is smaller than the reduced form.
\end{proof}

\begin{proposition}[GEWTR-2]
For $y$ in the eliminated sector, the form-domain criterion is
$\sup_{\delta>0}\langle y,(H_R+\delta)^{-1}y\rangle<\infty$ exactly when
$y\perp\ker H_R$ and $y\in\Dom H_{R,\rm red}^{-1/2}$.
\end{proposition}
\begin{proof}
Apply the spectral theorem and monotone convergence to
$\int(\lambda+\delta)^{-1}d\mu_y(\lambda)$.
\end{proof}
\begin{remark}
GEWTR begins with the global order $B\leq\lambda I$. It cannot prove that
order, locate $\lambda$, or supply a lower signed connected response.
\end{remark}

\begin{theorem}[CEOE/SDC-1 strict-tail equivalence]\label{thm:sdc}
Suppose $q>\sup\spec(E)$. Then
\[
 B\leq qI
 \quad\Longleftrightarrow\quad
 qP-A-C(qR-E)^{-1}D\geq0.
\]
At $q=\sup\spec(E)$, the equivalent form statement requires $X\geq0$,
$Ds\perp\ker H_R$, $Ds\in\Dom H_{R,\rm red}^{-1/2}$, and
\[
 \|H_{R,\rm red}^{-1/2}Ds\|^2\leq\langle s,Xs\rangle
\quad\text{for every retained }s.
\]
\end{theorem}
\begin{proof}
In the strict regime, triangular congruence factorization gives
\[
 qI-B=
 \begin{pmatrix}I&-CH_R^{-1}\\0&I\end{pmatrix}
 \begin{pmatrix}S_P(q)&0\\0&H_R\end{pmatrix}
 \begin{pmatrix}I&0\\-H_R^{-1}D&I\end{pmatrix}.
\]
Congruence preserves positivity and $H_R>0$, proving the equivalence. At the
touching edge, replace the inverse by the reduced closed quadratic form. The
positive-block contraction criterion is equivalent to the stated kernel,
domain, and form inequality.
\end{proof}
\begin{counterexample}[Bounded WTR does not establish a candidate edge]
Take scalar $A=q$, $E=0$, and $D=C=\varepsilon\neq0$. The tail resolvent
correction is finite, yet the top eigenvalue of the full $2\times2$ block is
strictly larger than $q$. Thus an upper WTR estimate cannot replace SDC-1.
\end{counterexample}
\begin{remark}
For a proposed physical edge $q_\beta=e^{-ma_\beta}<1$, the first open Wilson
lemma is precisely SDC-1 on the complete carrier, with uniform schedule and
norm control. The theorem repackages the target-strength order; it does not
make it true.
\end{remark}
